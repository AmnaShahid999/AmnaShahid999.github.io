# 3. DATA CLEANING AND PREPROCESSING
# =============================================================================
def clean_data(df):
    """Clean and preprocess the admission data"""
    df_clean = df.copy()

    # Ensure column names are stripped at the start of clean_data for consistency
    df_clean.columns = df_clean.columns.str.strip()

    # Handle missing values for key columns
    columns_to_fill = {
        'Application Fee Status' : 'No',
        'Gender': 'Unknown',
        'City-corrected': 'Unknown',
        'Area-corrected': 'Unknown',
        'Board': 'Unknown',
        'Took Test': 'No',
        'Appeared in Interview': 'No',
        'Admission Offered': 'No',
        'Finally Enrolled': 'No',
        'Admission Fee Status': 'No',
        'Intended School': 'Unknown',
        'Intended Program': 'Unknown',
        'Scholarship Status': 'None'
    }

    for col, default_value in columns_to_fill.items():
        if col in df_clean.columns:
            df_clean[col] = df_clean[col].fillna(default_value)

    # Standardize text values (convert to lowercase and strip whitespace)
    text_columns = ['Gender', 'City-corrected', 'Area-corrected', 'Board',
                   'Application Fee Status', 'Took Test', 'Appeared in Interview',
                   'Admission Offered', 'Finally Enrolled', 'Intended School',
                   'Intended Program', 'Scholarship Status']

    for col in text_columns:
        if col in df_clean.columns:
            df_clean[col] = df_clean[col].astype(str).str.strip()

    return df_clean

df_clean = clean_data(df) # Re-execute to create df_clean from the now cleaned df
print("Data cleaning completed!")

def encode_fee(x):
    if x.lower() == "Waived":
        return 1
    elif x.lower() in ["pending", "missing", ""]:
        return 0
    elif x.lower() == "submit":
        return 1
    else:
        return 1
